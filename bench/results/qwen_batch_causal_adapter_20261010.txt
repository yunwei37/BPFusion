def sliced_linear(self,x):
    if x.ndim==3 and x.shape[0]>1:
        calls['linear_slice']+=1
        return torch.cat([original_linear(self,x[i:i+1]) for i in range(x.shape[0])],dim=0)

def sliced_attention(module,query,key,value,attention_mask,scaling,dropout=0.0,**kwargs):
    if query.shape[0]==1:
        return original_attention(module,query,key,value,attention_mask,scaling,dropout,**kwargs)
    calls['attention_slice']+=1
    batch=query.shape[0];outputs=[];scores=[]
    assert key.shape[0]==value.shape[0]==batch
    assert attention_mask is None or attention_mask.shape[0] in (1,batch)
    for i in range(batch):
        mask=attention_mask if attention_mask is None or attention_mask.shape[0]==1 else attention_mask[i:i+1]
        output,score=original_attention(module,query[i:i+1],key[i:i+1],value[i:i+1],mask,scaling,dropout,**kwargs)
        outputs.append(output);scores.append(score)
    return torch.cat(outputs,dim=0),torch.cat(scores,dim=0)
