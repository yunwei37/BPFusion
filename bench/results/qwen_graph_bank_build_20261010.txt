struct ProbeControl { unsigned k,error,launches; };
__global__ void probe_dispatch(Step *steps,long long **inputs,unsigned *prompts,long long *results,unsigned count,unsigned banks,unsigned gen,ProbeControl *c) {
    if(threadIdx.x || blockIdx.x) return;
    if(c->k) {
        for(unsigned b=0;b<banks;b++) {
            unsigned previous=c->k==1 ? prompts[b] : BF_LLM_MAX_TOK+prompts[b]+c->k-2;
            long long token=*steps[b*count+previous].output;
            results[b*gen+c->k-1]=token;inputs[b][0]=token;
        }
    }
    if(c->k==gen) return;
    for(unsigned b=0;b<banks;b++) {
        unsigned index=c->k ? BF_LLM_MAX_TOK+prompts[b]+c->k-1 : prompts[b];
        cudaError_t e=cudaGraphLaunch(steps[b*count+index].graph,cudaStreamGraphFireAndForget);
        if(e!=cudaSuccess) { c->error=e;return; }
        c->launches++;
    }
    c->k++;
    cudaError_t e=cudaGraphLaunch(cudaGetCurrentGraphExec(),cudaStreamGraphTailLaunch);
    if(e!=cudaSuccess) c->error=e;
}
