// publish DONE after its last page access; TX alone never reclaims PENDING.

MODULE_PARM_DESC(reply_addr, "local server IPv4 address, network order");

static struct {
	struct sock *sk; /* lookup owns a reference, held until slot completion */
	u32 bytes;
	bool failed;
	u16 header_length, header_sent;
	char header[128];
} tx[BF_LLM_SLOTS];
static unsigned long sent_bytes, completions, abandoned, retries;

static int reply_tcp(u32 idx, struct bf_llm_slot *s, struct kvec *vec,
                     u32 nvec, u32 bytes, struct sock **locked)
{
	struct sock *sk = tx[idx].sk;
	struct msghdr msg = { .msg_flags = MSG_DONTWAIT | MSG_NOSIGNAL };
	int ret;

	if (!sk) {
		rcu_read_lock();
		sk = __inet_lookup_established(bf_net, (__be32)s->addr_be,
			(__be16)s->port_be, (__be32)reply_addr, BF_LLM_TCP_PORT, 0, 0);
		rcu_read_unlock();
		if (!sk)
			return -ENOENT;
		if (!sk_fullsock(sk)) {
			sock_gen_put(sk);
			return -ENOTCONN;
		}
		tx[idx].sk = sk;
	}
	iov_iter_kvec(&msg.msg_iter, ITER_SOURCE, vec, nvec, bytes);
	/* No RCU read section across lock_sock/send. No sk_socket dereference:
	 * close can detach that object even while a sock reference is held. */
	lock_sock(sk);
	*locked = sk;
	if (sock_flag(sk, SOCK_DEAD) || (sk->sk_shutdown & SEND_SHUTDOWN) ||
	    (sk->sk_state != TCP_ESTABLISHED && sk->sk_state != TCP_CLOSE_WAIT))
		ret = -ENOTCONN;
	else
		ret = tcp_sendmsg_locked(sk, &msg, bytes);
	/* Caller recycles a completed slot before releasing incoming backlog. */
	return ret;
}

/* Stream responses must retain request order on a persistent connection,
 * including ring wrap and a partially sent preceding response. */
static bool earlier_peer(u32 idx, struct bf_llm_slot *s)
{
    for (u32 i = 0; i < BF_LLM_SLOTS; i++) {
        struct bf_llm_slot *other = &page->llm[i];
        u32 state;

        if (i == idx)
            continue;
        state = smp_load_acquire(&other->state);
        if ((state == BF_PENDING || state == BF_DONE) &&
            other->addr_be == s->addr_be && other->port_be == s->port_be &&
            other->ingress_ns < s->ingress_ns)
            return true;
    }
    return false;
}

/* Binary replies have no status header. Reject by ending the owned peer,
 * so the client observes EOF and the rest of the GPU queue remains live. */
static void reject_binary(struct bf_llm_slot *s)
{
    struct peer *peer;

    list_for_each_entry(peer, &peers, node) {
        struct inet_sock *inet = inet_sk(peer->socket->sk);
        if ((__force u32)inet->inet_daddr == s->addr_be &&
            (__force u16)inet->inet_dport == s->port_be) {
            kernel_sock_shutdown(peer->socket, SHUT_RDWR);
            return;
        }
    }
}

static void poll_slot(u32 idx)
{
	struct bf_llm_slot *s = &page->llm[idx];
	struct sock *locked = NULL, *put = NULL;
	u32 state = smp_load_acquire(&s->state);
	u32 produced;
	u16 flags = READ_ONCE(s->pad), transport = flags & BF_LLM_TRANSPORT_MASK;
	bool header, rejected = flags & BF_LLM_REJECTED;
	int ret;

	if ((state != BF_PENDING && state != BF_DONE) ||
	    (transport != 1 && transport != 2))
		return;
	/* A bodyless rejection header also completes a response. */
	if ((flags & BF_LLM_VALIDATING) || (rejected && state != BF_DONE))
		return;
	if (earlier_peer(idx, s))
		return;
	produced = smp_load_acquire(&s->produced);
	/* Keep the last token until GPU release. Receiving a whole response must
	 * not permit a new request while this slot is still executor-owned. */
	if (state != BF_DONE && produced && produced == READ_ONCE(s->n_gen))
		produced--;
	if (produced > BF_LLM_MAX_TOK || (!rejected && READ_ONCE(s->n_gen) > BF_LLM_MAX_TOK))
		tx[idx].failed = true;
	if (rejected && transport == 1 && !tx[idx].failed) {
		reject_binary(s);
		tx[idx].failed = true;
	}
	if (transport == 2 && !tx[idx].header_length)
		tx[idx].header_length = scnprintf(tx[idx].header, sizeof(tx[idx].header),
			"HTTP/1.1 %s\r\nContent-Type: application/octet-stream\r\nContent-Length: %u\r\n\r\n",
			rejected ? "400 Bad Request" : "200 OK",
			rejected ? 0 : s->n_gen * (u32)sizeof(u32));
	header = tx[idx].header_sent < tx[idx].header_length;
	/* A valid HTTP header waits for the first available token so one send
	 * carries both. Bodyless rejection headers still complete immediately. */
	if (!tx[idx].failed && (!header || produced || rejected) &&
	    (header || produced * sizeof(u32) > tx[idx].bytes)) {
		struct kvec vec[2];
		u32 count = 0, header_bytes = 0, bytes = 0;
		if (header) {
			header_bytes = tx[idx].header_length - tx[idx].header_sent;
			vec[count++] = (struct kvec){ tx[idx].header + tx[idx].header_sent, header_bytes };
			bytes += header_bytes;
		}
		if (produced * sizeof(u32) > tx[idx].bytes) {
			u32 body_bytes = produced * sizeof(u32) - tx[idx].bytes;
			vec[count++] = (struct kvec){ (u8 *)s->tok_out + tx[idx].bytes, body_bytes };
			bytes += body_bytes;
		}
		ret = reply_tcp(idx, s, vec, count, bytes, &locked);
		if (ret > 0) {
			u32 sent_header = min_t(u32, ret, header_bytes);
			tx[idx].header_sent += sent_header;
			tx[idx].bytes += ret - sent_header; /* includes a partial u32 */
			sent_bytes += ret;
		} else {
			retries++;
			if (ret == -EPIPE || ret == -ECONNRESET || ret == -ENOTCONN ||
			    (ret == -ENOENT && state == BF_DONE))
				tx[idx].failed = true;
		}
	}
	/* A disconnected peer cannot release memory still owned by an executor.
	 * DONE is its release; produced == n_gen is not a release. */
	if (state == BF_DONE && (tx[idx].failed ||
	    (tx[idx].header_sent == tx[idx].header_length &&
	    tx[idx].bytes == produced * sizeof(u32)))) {
		put = tx[idx].sk;
		if (tx[idx].failed)
			abandoned++;
		else
			completions++;
		memset(&tx[idx], 0, sizeof(tx[idx]));
		smp_store_release(&s->state, BF_FREE);
	}
	/* release_sock can run the next request's parser from TCP backlog. All
	 * old slot/TX accesses must finish first; keep the sock reference until
	 * after unlocking, even when completion cleared tx[idx].sk. */
	if (locked)
		release_sock(locked);
	if (put)
		sock_put(put);
}

/* The bootstrap process owns the listener file; sockfd_lookup pins it.
 * Accepted sockets and receive-buffer draining are entirely kernel-owned. */
static bool peer_pending(struct sock *sk)
{
	struct inet_sock *inet = inet_sk(sk);

	for (u32 i = 0; i < BF_LLM_SLOTS; i++) {
		struct bf_llm_slot *s = &page->llm[i];
		u32 state = smp_load_acquire(&s->state);

		if (state != BF_FREE && ((s->pad & BF_LLM_TRANSPORT_MASK) == 1 ||
                                  (s->pad & BF_LLM_TRANSPORT_MASK) == 2) &&
		    s->addr_be == (__force u32)inet->inet_daddr &&
		    s->port_be == (__force u16)inet->inet_dport)
			return true;
	}
	return false;
}

static void accept_and_drain(void)
{
	struct socket *socket;
	struct peer *peer, *next;

	if (!listener)
		return;
	while (kernel_accept(listener, &socket, O_NONBLOCK) == 0) {
		peer = kmalloc(sizeof(*peer), GFP_KERNEL);
		if (!peer) {
			sock_release(socket);
			continue;
		}
		peer->socket = socket;
		list_add_tail(&peer->node, &peers);
		accepted++;
	}
	list_for_each_entry_safe(peer, next, &peers, node) {
		char discard[256];
		struct kvec vec = { .iov_base = discard, .iov_len = sizeof(discard) };
		int ret;

		do {
			struct msghdr msg = {};

			ret = kernel_recvmsg(peer->socket, &msg, &vec, 1,
					     sizeof(discard), MSG_DONTWAIT);
		} while (ret > 0);
		if ((ret == 0 && !peer_pending(peer->socket->sk)) ||
		    (ret < 0 && ret != -EAGAIN && ret != -EWOULDBLOCK)) {
			list_del(&peer->node);
			sock_release(peer->socket);
			kfree(peer);
			closed++;
		}
	}
}

static int tx_thread(void *unused)
{
	while (!kthread_should_stop()) {
		accept_and_drain();
		for (u32 i = 0; i < BF_LLM_SLOTS; i++)
			poll_slot(i);
		usleep_range(60, 120);
	}
	return 0;
}

static int __init bf_init(void)
{
	struct bpf_array *array;
	int err;

	/* bpf_map_get validates the actual BPF FD and acquires a map reference.
	 * A bpffs filp_open FD is a seq_file, not a BPF map FD. */
	page_map = bpf_map_get(map_fd);
	if (IS_ERR(page_map))
		return PTR_ERR(page_map);
	if (page_map->map_type != BPF_MAP_TYPE_ARRAY ||
	    !(page_map->map_flags & BPF_F_MMAPABLE) ||
	    page_map->max_entries != 1 || page_map->value_size != sizeof(*page)) {
		err = -EINVAL;
		goto out_map;
	}
	array = container_of(page_map, struct bpf_array, map);
	page = (struct bf_page *)array->value;
	bf_net = get_net(current->nsproxy->net_ns);
	if (listen_fd >= 0) {
		listener = sockfd_lookup(listen_fd, &err);
		if (!listener)
			goto out_net;
		if (listener->type != SOCK_STREAM || listener->sk->sk_protocol != IPPROTO_TCP ||
		    listener->sk->sk_state != TCP_LISTEN ||
		    inet_sk(listener->sk)->inet_num != BF_LLM_TCP_PORT ||
		    !net_eq(sock_net(listener->sk), bf_net)) {
			err = -EINVAL;
			goto out_listener;
		}
	}
	tx_task = kthread_run(tx_thread, NULL, "bfusion_tx");
	if (IS_ERR(tx_task)) {
		err = PTR_ERR(tx_task);
		goto out_listener;
	}
	pr_info("bfusion_tx: map id=%u size=%u ring=%zu slot=%zu net=%u\n",
		page_map->id, page_map->value_size, offsetof(struct bf_page, llm),
		sizeof(struct bf_llm_slot), bf_net->ns.inum);
	return 0;
out_listener:
	if (listener)
		sockfd_put(listener);
out_net:
	put_net(bf_net);
out_map:
	bpf_map_put(page_map);
	return err;
}

static void __exit bf_exit(void)
{
	struct peer *peer, *next;

	kthread_stop(tx_task);
	list_for_each_entry_safe(peer, next, &peers, node) {
		list_del(&peer->node);
		sock_release(peer->socket);
		kfree(peer);
	}
	if (listener)
		sockfd_put(listener);
	for (u32 i = 0; i < BF_LLM_SLOTS; i++)
		if (tx[i].sk)
			sock_put(tx[i].sk);
	bpf_map_put(page_map);
	put_net(bf_net);
	pr_info("bfusion_tx: exit bytes=%lu completed=%lu abandoned=%lu retries=%lu accepted=%lu closed=%lu\n",
		sent_bytes, completions, abandoned, retries, accepted, closed);
}
module_init(bf_init);
module_exit(bf_exit);

static int user_network(bf_page *page,int listener,unsigned seconds,
                        volatile sig_atomic_t *stopped) {
    struct Peer {
        int fd;
        bool eof=false;
        int slot=-1;
        std::string input;
    };
    struct Reply {
        std::shared_ptr<Peer> peer;
        std::string header;
        size_t header_sent=0,bytes=0;
    } replies[BF_LLM_SLOTS];
    std::map<int,std::shared_ptr<Peer>> peers;
    unsigned head=0,completed=0,rejected_count=0,accepted=0;
    int ep=epoll_create1(EPOLL_CLOEXEC);
    if(ep<0) { perror("epoll_create1");return 1; }
    fcntl(listener,F_SETFL,fcntl(listener,F_GETFL)|O_NONBLOCK);
    epoll_event watch={};watch.events=EPOLLIN;watch.data.fd=listener;
    if(epoll_ctl(ep,EPOLL_CTL_ADD,listener,&watch)) { close(ep);return 1; }
    auto disconnect=[&](Peer &peer) {
        if(peer.fd>=0) {
            epoll_ctl(ep,EPOLL_CTL_DEL,peer.fd,nullptr);
            close(peer.fd);peer.fd=-1;
        }
        peer.eof=true;
    };
    auto send_reply=[&](Peer &peer,Reply &reply,bf_llm_slot *slot,unsigned produced) {
        iovec vec[2];unsigned count=0;
        size_t header_bytes=reply.header.size()-reply.header_sent;
        if(header_bytes) vec[count++]={reply.header.data()+reply.header_sent,header_bytes};
        if(produced*4>reply.bytes)
            vec[count++]={(char *)slot->tok_out+reply.bytes,produced*4-reply.bytes};
        if(!count) return;
        msghdr msg={};msg.msg_iov=vec;msg.msg_iovlen=count;
        ssize_t n=sendmsg(peer.fd,&msg,MSG_DONTWAIT|MSG_NOSIGNAL);
        if(n<0 && errno!=EAGAIN && errno!=EWOULDBLOCK && errno!=EINTR)
            disconnect(peer);
        if(n>0) {
            size_t sent_header=(size_t)n<header_bytes ? (size_t)n : header_bytes;
            reply.header_sent+=sent_header;reply.bytes+=(size_t)n-sent_header;
        }
    };
    auto deadline=std::chrono::steady_clock::now()+std::chrono::seconds(seconds);
    while(!*stopped && std::chrono::steady_clock::now()<deadline) {
        // Busy completion polling is a low-latency control; report its CPU cost.
        epoll_event events[BF_LLM_SLOTS*2];
        int n=epoll_wait(ep,events,BF_LLM_SLOTS*2,0);
        if(n<0 && errno!=EINTR) { perror("epoll_wait");break; }
        for(int e=0;e<n;e++) {
            int fd=events[e].data.fd;
            if(fd==listener) {
                while(true) {
                    int child=accept4(listener,nullptr,nullptr,SOCK_NONBLOCK|SOCK_CLOEXEC);
                    if(child<0) break;
                    auto peer=std::make_shared<Peer>();peer->fd=child;
                    peers[child]=peer;accepted++;
                    watch.events=EPOLLIN|EPOLLRDHUP;watch.data.fd=child;
                    if(epoll_ctl(ep,EPOLL_CTL_ADD,child,&watch)) disconnect(*peer);
                }
            } else {
                auto it=peers.find(fd);if(it==peers.end()) continue;
                Peer &peer=*it->second;
                char buffer[4096];
                while(peer.fd>=0) {
                    ssize_t got=recv(peer.fd,buffer,sizeof(buffer),MSG_DONTWAIT);
                    if(got>0) peer.input.append(buffer,got);
                    else {
                        if(got==0) {
                            peer.eof=true;epoll_ctl(ep,EPOLL_CTL_DEL,peer.fd,nullptr);
                        } else if(errno!=EAGAIN && errno!=EWOULDBLOCK && errno!=EINTR)
                            disconnect(peer);
                        break;
                    }
                }
            }
        }
        for(unsigned i=0;i<BF_LLM_SLOTS;i++) {
            Reply &reply=replies[i];if(!reply.peer) continue;
            Peer &peer=*reply.peer;bf_llm_slot *slot=&page->llm[i];
            unsigned state=__atomic_load_n(&slot->state,__ATOMIC_ACQUIRE);
            unsigned short flags=__atomic_load_n(&slot->pad,__ATOMIC_ACQUIRE);
            bool rejected=flags&BF_LLM_REJECTED;
            if((flags&BF_LLM_VALIDATING) || (rejected && state!=BF_DONE)) continue;
            unsigned produced=__atomic_load_n(&slot->produced,__ATOMIC_ACQUIRE);
            if(state!=BF_DONE && produced==slot->n_gen && produced) produced--;
            if(rejected && (flags&BF_LLM_TRANSPORT_MASK)==1) disconnect(peer);
            if((flags&BF_LLM_TRANSPORT_MASK)==2 && reply.header.empty()) {
                reply.header="HTTP/1.1 "+std::string(rejected ? "400 Bad Request" : "200 OK")
                    +"\r\nContent-Type: application/octet-stream\r\nContent-Length: "
                    +std::to_string(rejected ? 0 : slot->n_gen*4)+"\r\n\r\n";
            }
            // Keep default TCP buffering; submit header and first tokens together.
            if(peer.fd>=0 && (produced || rejected))
                send_reply(peer,reply,slot,produced);
            if(state==BF_DONE && (peer.fd<0 ||
                (reply.header_sent==reply.header.size() && reply.bytes==produced*4))) {
                peer.slot=-1;reply={};
                __atomic_store_n(&slot->state,(unsigned)BF_FREE,__ATOMIC_RELEASE);
                completed++;if(rejected) rejected_count++;
            }
        }
        for(auto it=peers.begin();it!=peers.end();) {
            auto peer=it->second;
            if(peer->fd>=0 && peer->slot<0 && !peer->input.empty()) {
                size_t offset=0;
                bool http=peer->input.compare(0,4,"POST")==0;
                if(http) {
                    size_t end=peer->input.find("\r\n\r\n");
                    if(end==std::string::npos) { ++it;continue; }
                    offset=end+4;
                }
                if(peer->input.size()<offset+24) { ++it;continue; }
                unsigned fields[4];unsigned long long client_ns;
                memcpy(fields,peer->input.data()+offset,16);
                memcpy(&client_ns,peer->input.data()+offset+16,8);
                unsigned prompt=fields[1],gen=fields[2];
                if(fields[0]!=BF_LLM_MAGIC || !prompt || prompt>BF_LLM_MAX_TOK ||
                   !gen || gen>BF_LLM_MAX_TOK) {
                    disconnect(*peer);
                } else {
                    size_t bytes=24+prompt*4;
                    if(http) {
                        size_t key=peer->input.find("\r\nContent-Length: ");
                        if(key==std::string::npos || key>=offset ||
                           strtoul(peer->input.c_str()+key+18,nullptr,10)!=bytes) {
                            disconnect(*peer);++it;continue;
                        }
                    }
                    if(peer->input.size()<offset+bytes) { ++it;continue; }
                    unsigned i=head%BF_LLM_SLOTS;bf_llm_slot *slot=&page->llm[i];
                    if(__atomic_load_n(&slot->state,__ATOMIC_ACQUIRE)==BF_FREE) {
                        __atomic_store_n(&slot->state,(unsigned)BF_WRITING,__ATOMIC_RELEASE);
                        slot->n_prompt=prompt;slot->n_gen=gen;slot->produced=0;
                        slot->client_ns=client_ns;
                        slot->pad=(http ? 2 : 1)|BF_LLM_VALIDATING;
                        memcpy(slot->tok_in,peer->input.data()+offset+24,prompt*4);
                        peer->input.erase(0,offset+bytes);
                        replies[i].peer=peer;peer->slot=i;
                        __atomic_store_n(&slot->state,(unsigned)BF_PENDING,__ATOMIC_RELEASE);
                        __atomic_store_n(&page->llm_head,++head,__ATOMIC_RELEASE);
                    }
                }
            }
            if(peer->slot<0 && (peer->fd<0 || (peer->eof && peer->input.empty()))) {
                disconnect(*peer);it=peers.erase(it);
            } else ++it;
        }
    }
    for(auto &item:peers) disconnect(*item.second);
    close(ep);
    printf("userspace TCP stopped: accepted=%u published=%u completed=%u rejected=%u\n",
           accepted,head,completed,rejected_count);fflush(stdout);
    return 0;
}

struct Step { cudaGraphExec_t graph; long long *output; };
struct Request { unsigned k,prompt,gen,step,phase; };
struct Control { unsigned seen,retired,reclaim,error,launches,max_active; Request requests[BF_LLM_SLOTS]; };

// System-scope acquire protects host publication and forbids polling-load
// elimination; a cache modifier alone does not give synchronization semantics.
__device__ unsigned queue_load(const unsigned *p) {
    unsigned value;asm volatile("ld.acquire.sys.global.u32 %0, [%1];" : "=r"(value) : "l"(p) : "memory");return value;
}
__device__ unsigned long long queue_load(const unsigned long long *p) {
    unsigned long long value;asm volatile("ld.acquire.sys.global.u64 %0, [%1];" : "=l"(value) : "l"(p) : "memory");return value;
}

__global__ void dispatch(bf_page *p,Step *steps,long long **inputs,unsigned count,unsigned vocab,Control *c,bool once) {
    if(threadIdx.x || blockIdx.x) return;
    while(!queue_load((unsigned long long *)&p->stop_ns)) {
        // A parent tail resumes only after all private child graphs finish.
        // phase1 awaits sampling; phase3 awaits launch, including after a
        // host-control return that publishes another request's DONE.
        for(unsigned i=0;i<BF_LLM_SLOTS;i++) {
            Request *r=&c->requests[i];
            if(r->phase!=1) continue;
            unsigned bank=i;
            unsigned token=(unsigned)*steps[bank*count+r->step].output;
            bf_llm_slot *slot=&p->llm[i];
            __stcg(&slot->tok_out[r->k],token);__threadfence_system();__stcg(&slot->produced,r->k+1);
            if(++r->k==r->gen) r->phase=2;
            else { inputs[bank][0]=token;r->step=BF_LLM_MAX_TOK+r->prompt+r->k-1;r->phase=3; }
        }
        // The head-indexed ring must reclaim in admission order. Hold final
        // tokens until the previous DONE was reclaimed by kernel TX. A reused
        // WRITING/PENDING slot also proves that its old ownership was released.
        if(c->reclaim && queue_load(&p->llm[(c->retired-1)%BF_LLM_SLOTS].state)!=BF_DONE) c->reclaim=0;
        unsigned idx=c->retired%BF_LLM_SLOTS;
        if(!c->reclaim && c->retired!=c->seen && c->requests[idx].phase==2) {
            __threadfence_system();__stcg(&p->llm[idx].state,(unsigned)BF_DONE);
            c->requests[idx].phase=0;c->retired++;c->reclaim=1;
            if(once) return;
        }
        // Admission remains sequential; computation uses each physical slot's
        // private KV/input/output bank. No producer or page ABI change.
        while(queue_load(&p->llm_head)!=c->seen) {
            idx=c->seen%BF_LLM_SLOTS;
            bf_llm_slot *slot=&p->llm[idx];
            if(c->requests[idx].phase || queue_load(&slot->state)!=BF_PENDING) break;
            unsigned bank=idx;
            unsigned np=__ldcg(&slot->n_prompt),ng=__ldcg(&slot->n_gen);
            bool bad=!np || np>BF_LLM_MAX_TOK || !ng || ng>BF_LLM_MAX_TOK;
            for(unsigned j=0;j<np && j<BF_LLM_MAX_TOK;j++) {
                unsigned token=__ldcg(&slot->tok_in[j]);
                if(token>=vocab) bad=true;
                inputs[bank][j]=token;
            }
            unsigned short flags=__ldcg(&slot->pad);
            __threadfence_system();
            __stcg(&slot->pad,(unsigned short)((flags&~BF_LLM_VALIDATING)|(bad ? BF_LLM_REJECTED : 0)));
            c->requests[idx]={0,np,ng,np,bad ? 2u : 3u};c->seen++;
        }
        unsigned active=0;
        for(unsigned i=0;i<BF_LLM_SLOTS;i++) {
            Request *r=&c->requests[i];
            if(r->phase!=3) continue;
            unsigned bank=i;
            cudaError_t e=cudaGraphLaunch(steps[bank*count+r->step].graph,cudaStreamGraphFireAndForget);
            if(e!=cudaSuccess) { c->error=e;return; }
            r->phase=1;c->launches++;active++;
        }
        if(active) {
            if(active>c->max_active) c->max_active=active;
            cudaError_t e=cudaGraphLaunch(cudaGetCurrentGraphExec(),cudaStreamGraphTailLaunch);
            if(e!=cudaSuccess) c->error=e;
            return;
        }
        // Idle, rejection and TX reclamation waits stay on the GPU.
    }
}

def request(client,ids,expected):
    body=struct.pack("<IIIIQ",0x514c4d51,len(ids),len(expected),0,time.monotonic_ns())+struct.pack(f"<{len(ids)}I",*ids)
    wire=f"POST /generate HTTP/1.1\r\nHost: localhost\r\nContent-Length: {len(body)}\r\n\r\n".encode()+body
    start=time.monotonic_ns()
    client.sendall(wire)
    header=b""; data=b""; arrivals=[]
    while len(data)<4*len(expected):
        chunk=client.recv(4096)
        arrived=time.monotonic_ns()
        assert chunk,"short reply"
        if header is not None:
            header+=chunk
            if b"\r\n\r\n" not in header: continue
            head,chunk=header.split(b"\r\n\r\n",1)
            assert head.startswith(b"HTTP/1.1 200 OK\r\n"),head
            assert f"Content-Length: {4*len(expected)}".encode() in head,head
            header=None
        data+=chunk
        arrivals.extend([arrived]*((len(data)//4)-len(arrivals)))
    assert len(data)==4*len(expected),data
    actual=list(struct.unpack(f"<{len(expected)}I",data))
    assert actual==expected,(ids,actual,expected)
    return {"start_ns":start,"arrival_ns":arrivals,"tokens":actual,
            "ttft_ms":(arrivals[0]-start)/1e6,
            "latency_ms":(arrivals[-1]-start)/1e6,
            "tpot_ms":(arrivals[-1]-arrivals[0])/(len(expected)-1)/1e6 if len(expected)>1 else None}

def cell(pid,cases,gen,concurrency,count,rows):
    # Connections and thread creation precede both the timer and CPU window.
    barrier=threading.Barrier(concurrency+1)
    ready=threading.Barrier(concurrency+1)
    def worker(worker_id):
        with connect() as client:
            ready.wait(timeout=60)
            barrier.wait(timeout=60)
            for index in range(worker_id,count,concurrency):
                case=index%len(cases); ids,expected=cases[case]
                row=request(client,ids,expected[:gen]); row.update(index=index,case=case)
                rows.append(row)
    with ThreadPoolExecutor(max_workers=concurrency) as pool:
        futures=[pool.submit(worker,i) for i in range(concurrency)]
        ready.wait(timeout=60)
        before=cpu_seconds(pid); host_before=host_cpu_ticks(); start=time.monotonic_ns()
        barrier.wait(timeout=60)
        for future in futures: future.result()
        finish=time.monotonic_ns(); host_after=host_cpu_ticks(); after=cpu_seconds(pid)
    first=min(row["start_ns"] for row in rows)
    last=max(row["arrival_ns"][-1] for row in rows)
    return {
        "window_ns":finish-start,"executor_cpu_s":after-before,
        "host_cpu_ticks_before":host_before,"host_cpu_ticks_after":host_after,
        "host_cpu_work_s":sum(host_after[i]-host_before[i] for i in (0,1,2,5,6))/os.sysconf("SC_CLK_TCK"),
        "output_tokens_per_s":count*gen/((last-first)/1e9)}

def oracle_cases(gen, prefill_edges=False, name="Qwen/Qwen2.5-0.5B-Instruct"):
    tok=AutoTokenizer.from_pretrained(name)
    model=AutoModelForCausalLM.from_pretrained(name,dtype=torch.float16,attn_implementation="eager").to("cuda").eval()
    vocab=model.config.vocab_size
    print(f"HF reference model={name} type={model.config.model_type} vocab={vocab}",flush=True)
    cases=[]
    inputs=[tok(prompt).input_ids for prompt in ("The capital of France is", "What is 2 plus 2?", "Write a short greeting.", "Linux is")]
    if prefill_edges:
        pattern=inputs[0]
        inputs.extend((pattern*((n+len(pattern)-1)//len(pattern)))[:n] for n in (1,15,16,17,32,64))
    for ids in inputs:
        with torch.no_grad():
            out=model(torch.tensor([ids],device="cuda"),use_cache=True)
            past=out.past_key_values
            nxt=int(out.logits[:,-1].argmax(-1))
            expected=[nxt]
            for _ in range(gen-1):
                out=model(torch.tensor([[nxt]],device="cuda"),past_key_values=past,use_cache=True)
                past=out.past_key_values
                nxt=int(out.logits[:,-1].argmax(-1))
                expected.append(nxt)
        cases.append((ids,expected))
    del model,out,past
    gc.collect(); torch.cuda.empty_cache()
    return cases,vocab
