  assert generated==c['expected'][:index+1], (case,generated,c['expected'][:index+1])

  delta=plow-ref
  result={'case':case,'failed_output_index':index,'hf_generate_prefix':generated,'fixed_oracle_prefix_match':True,'hf_raw_logit_dtype':str(output.logits[index].dtype),'hf_generate_top10':top(ref),'plow_top10':top(plow),'hf_argmax':int(ref.argmax()),'plow_argmax':int(plow.argmax()),'reference_top2_margin':float(ref.topk(2).values.diff().abs()[0]),'n_logits':ref.numel(),'all_finite':bool(torch.isfinite(ref).all() and torch.isfinite(plow).all()),'max_abs':float(delta.abs().max()),'mean_abs':float(delta.abs().mean()),'relative_l2':float(torch.linalg.vector_norm(delta)/torch.linalg.vector_norm(ref)),'kl_ref_to_plow':float((ref.softmax(0)*(ref.log_softmax(0)-plow.log_softmax(0))).sum())}
