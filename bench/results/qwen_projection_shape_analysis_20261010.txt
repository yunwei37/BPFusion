from pathlib import Path
import json,gzip,collections,re
base=Path('bench/results')
raw=(base/'qwen_projection_shape_full_20261010.txt').read_text()
assert raw.endswith('EXIT 0\n')
rows=[json.loads(l[len('SHAPE_RESULT '):]) for l in raw.splitlines() if l.startswith('SHAPE_RESULT ')]
assert [r['index'] for r in rows]==[5,64,80,128,190]
for row in rows:
    assert row['eager_token']==row['captured_token']
    assert row['projection_names']==169
    assert not row['unmatched_cuda_records'] and not row['extra_association_records']
    traces=base/('qwen_projection_shape_'+str(row['index'])+'_20261010.json.gz')
    with gzip.open(traces,'rt') as f:t=json.load(f)
    kernels=[e for e in t['traceEvents'] if e.get('cat') in ('kernel','gpu_memcpy','gpu_memset')]
    exported=collections.Counter((e['name'],round(e['dur'],6)) for e in kernels)
    observed=collections.Counter((e['name'],round(e['duration_us'],6)) for e in row['cuda_records'])
    assert exported==observed,(exported-observed,observed-exported)
    grouped=collections.defaultdict(lambda:{'calls':0,'kernels':0,'device_us':0.0,'shapes':set(),'kernel_names':collections.Counter()})
    for op in row['operations']:
        if op['projection']:
            family=re.sub(r'\.\d+\.', '.L.',op['projection'])
        else:
            family='UNLABELLED:'+op['operation']
        item=grouped[family];item['calls']+=1;item['shapes'].add(json.dumps(op['input_shapes']))
        for k in op['kernels']:
            item['kernels']+=1;item['device_us']+=k['duration_us'];item['kernel_names'][k['name']]+=1
    total=sum(e['duration_us'] for e in row['cuda_records'])
    assert abs(sum(g['device_us'] for g in grouped.values())-total)<1e-4
    summary=[]
    for family,item in grouped.items():
        item['shapes']=sorted(item['shapes']);item['kernel_names']=dict(item['kernel_names'])
        item['share_of_summed_kernel_duration']=item['device_us']/total
        summary.append({'family':family,**item})
    summary.sort(key=lambda x:x['device_us'],reverse=True)
    print(json.dumps({'index':row['index'],'count':row['count'],'position':row['position'],
          'all_cuda_records':len(row['cuda_records']),'associated_kernel_records':row['associated_kernel_records'],
          'summed_kernel_us':total,'coverage':'all directly associated and byte-export numerical counts match',
          'grouped':summary,'scope':'profiled eager, shared GPU; sums are kernel durations, not TCP/device-tail wall-time fractions'}),flush=True)
print('PROJECTION_ANALYSIS_PASS',flush=True)
